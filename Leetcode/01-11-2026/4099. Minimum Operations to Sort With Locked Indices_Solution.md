# 4099. Minimum Operations to Sort With Locked Indices

**LeetCode:** [4099. Minimum Operations to Sort With Locked Indices](https://leetcode.com/problems/minimum-operations-to-sort-with-locked-indices/description/)
    

## Solution
    
```python
    class Solution:
    def minOperations(self, nums: list[int], locked: list[int], k: int) -> int:

        n = len(nums)

        track = {}

        def check(i,e):
            return e == i+1
        locked_oo = 0
        for i,e in enumerate(nums):
            if e not in track:
                track[e] = 0

            track[e] += i
            if locked[i] and not check(i,e):
                locked_oo += 1

        ans = 0
        if locked_oo > k:
            return -1

        for e in range(1,n+1):
            if not check(track[e],e):
                
                ## unlock before it
                if sum(locked[e-1:track[e]+1]):
                    if k > 0:
                        # print(e,k)
                        for j in range(e-1,track[e]+1):
                            if locked[j]:
                                locked[j] = 0
                                ans += 1
                                k -= 1
                                # print(e,k)
                            if k < 0 :
                                return -1
                    else:
                        return -1
                # print(e,locked)
                # print(track)
                
                ## handle rest index
                for K,v in track.items():
                    if v < track[e] and v >= e-1:
                        track[K] += 1
                ans += (track[e]-(e-1))
                track[e] = e-1

        return ans
