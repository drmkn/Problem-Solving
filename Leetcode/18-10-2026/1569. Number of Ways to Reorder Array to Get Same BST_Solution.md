    class node:
    def __init__(self, value):
        self.left = None
        self.right = None
        self.value = value


class Solution:
    def numOfWays(self, nums: list[int]) -> int:
        n = len(nums)
        M = 10**9 +7

        root = node(nums[0])

        for e in nums[1:]:
            Node = root
            while Node:
                prev = Node
                if Node.value > e:
                    Node = Node.left
                else:
                    Node = Node.right
            if prev.value > e:
                prev.left = node(e)
            else:
                prev.right = node(e)

        def nCr(n, r):
            if r < 0 or r > n:
                return 0

            r = min(r, n - r)  # Use symmetry

            result = 1
            for i in range(1, r + 1):
                result = result * (n - i + 1) // i

            return result

        def f(node,count):
