# 4085. Find Root of a Colored Tree

**LeetCode:** [4085. Find Root of a Colored Tree](https://leetcode.com/problems/find-root-of-a-colored-tree/description/)
    

## Solution
    
```python
    class Solution:
    def findColoredRoot(self, edges: list[list[int]], colors: str) -> int:
        ## passed 562/663 cases
        n = len(colors)
        m = len(edges)

        degree_h = {}
        adj = {}
        for e in edges:
            if e[0] not in degree_h:
               degree_h[e[0]] = 0 
            degree_h[e[0]] += 1
            if e[0] not in adj:
               adj[e[0]] = [] 
            adj[e[0]].append(e[1])
            if e[1] not in degree_h:
               degree_h[e[1]] = 0 
            degree_h[e[1]] += 1
            if e[1] not in adj:
               adj[e[1]] = [] 
            adj[e[1]].append(e[0])

        # print(adj)    

        choices = []
        for k,v in degree_h.items():
            if v  < 3:
                choices.append(k)
                
        color_h = {'R':'B','B':'W','W':'R'}

        def f(node):
            print(node,visited)
            for nbr in adj[node]:
                if nbr not in visited:
                    visited.append(nbr)
                    if color_h.get(colors[node]) == colors[nbr]:
                        f(nbr)   
                    else:
                        return False
            if len(visited) == n:
                return True

            return False
    
        for node in choices:
            visited = [node]
            if f(node=node):
                return node
            
            
        return -1
