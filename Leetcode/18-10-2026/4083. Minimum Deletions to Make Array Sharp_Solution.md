    class Solution:
    def minDeletions(self, nums: list[int]) -> int:
        ## Weekly contest 524
        n = len(nums)

        l = []

        for i in range(n-1):
            if nums[i]^nums[i+1] == 0:
                l.append(i)

        # for j in l:
