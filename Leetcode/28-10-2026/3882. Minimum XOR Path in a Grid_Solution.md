        m, n = len(grid), len(grid[0])
        K = 1024

        dp = [[[False] * K for _ in range(n)] for _ in range(m)]
        dp[0][0][grid[0][0]] = True

        for i in range(m):
            for j in range(n):
                cur = dp[i][j]

                # Move down
                if i + 1 < m:
                    nxt = dp[i + 1][j]
                    val = grid[i + 1][j]

                        if cur[x]:
