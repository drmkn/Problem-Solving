# 3886. Sum of Sortable Integers

**LeetCode:** [3886. Sum of Sortable Integers](https://leetcode.com/problems/sum-of-sortable-integers/description/?envType=daily-question&envId=2026-10-20)
    

## Solution
    
```python
    from collections import Counter
class Solution:
    def sortableIntegers(self, nums: list[int]) -> int:
        n = len(nums)

        def check(a):
            m = len(a)
            dip = 0
            for i in range(1,m+1):
                if a[(i-1)%m] > a[(i)%m]:
                    dip += 1
                if dip >= 2:
                    return False
            
            return True
        
        def check_l(a,b):
            return Counter(a) == Counter(b)

        
        # print(check([7,6,5]))
        # print(Counter([1,2,2,3,2,3,3,4]) == Counter([4,2,2,3,3,2,3,1]))

        # for i in range(0,6,2):
        #     print(i)
        # print(i)

        ans = 0
        nums_sorted = sorted(nums)

        all_flag = 0
        if nums_sorted == nums:
            all_flag = 1

        for i in range(n):

            if not n%(i+1):
                if not all_flag:
                    flag = 0
                    for s in range(0,n,i+1):
                        # l = nums[s:s+i+1]
                        # print(nums[s:s+i+1],nums_sorted[s:s+i+1])
                        if check_l(nums[s:s+i+1],nums_sorted[s:s+i+1]):
                            if check(nums[s:s+i+1]):
                                continue
                            else:
                                flag = 1
                        else:
                            flag = 1
                        if flag:
                            break
                    if not flag:
                        ans += (i+1) 
                else:
                    ans += (i+1)



        return  ans
