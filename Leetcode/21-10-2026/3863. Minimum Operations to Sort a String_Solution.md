    class Solution:
    def minOperations(self, s: str) -> int:
        mn,mx = 0,0
        n = len(s)
        sorted_s = "".join(sorted(s))

        if sorted_s == s:
            return 0

        if n == 2:
            return -1

        for i,e in enumerate(s):
            if s[mn] > e:
                mn = i
            if s[mx] <= e:
                mx = i
        # min_single,max_single = True,True
        # indices_min = [index for index, char in enumerate(s) if char == s[mn]]
        # indices_max = [index for index, char in enumerate(s) if char == s[mx]]
        # if len(indices_min) > 1:
        #     min_single = False
        # if len(indices_max) > 1:
        #     max_single = False
        
        
        if mn == 0 or mx == n-1:
            return 1
        elif mn == n-1 and mx == 0:
            return 3
        else:
            return 2
