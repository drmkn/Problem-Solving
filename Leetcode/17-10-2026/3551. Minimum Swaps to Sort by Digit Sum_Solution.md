        n = len(nums)
        '''
        [0,1,2,3,4]
        [1,2,0,3,4]
        pi(0) = 1
        pi(1) = 2
        pi(2) = 0 
        pi(3) = 3
        pi(4) = 4

        pi = (012)(3)(4)
        c_k -> length k -> k-1 swaps
        swaps = sum of swaps needed for each c_k

                = sum(k-1) = N-#cycles
        '''
        def sum_digit(e):
            sum = 0
            while e:
                sum += e%10
                e //= 10
            # print(sum)
            return sum

            H[e] = sum_digit(e)
        # ans =  sorted(nums)
        # ans = sorted(nums,key= lambda x : (sum_digit(x),x))
        ans  = sorted(nums,key = lambda x : (H[x],x))
        h = {}
        for i,e in enumerate(nums):
            h[e] = i
        

        ordered_indices = [h[e] for e in ans]

                a,b = i,ordered_indices[i]
                ordered_indices[a],ordered_indices[b] = ordered_indices[b],ordered_indices[a] 
                s += 1
        
        return s
